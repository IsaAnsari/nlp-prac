'''
Part-of-speech (POS) tagging in NLP is the process of automatically assigning 
grammatical tags (like nouns, verbs, or adjectives) to each word in a sentence.

In Natural Language Processing (NLP), HMMs are primarily applied to sequential 
tagging and labeling tasks.

Neural models for Part-of-Speech (POS) tagging use deep learning architectures 
to automatically predict the grammatical label of words based on their surrounding context.

python -m spacy download en_core_web_sm
OR
py -[version] -m pip install spacy

'''

import spacy
import nltk

nltk.download('averaged_perceptron_tragger')
nltk.download('punkt_tab')
nltk.download('averaged_perceptron_tragger_eng')

text = "I love NLP Projects"
pos_tags = nltk.pos_tag(nltk.word_tokenize(text))

nlp = spacy.load("en_core_web_sm")
doc = nlp(text)
spacy_tags = [(token.text, token.pos_) for token in doc]

print("HMM Tags",pos_tags)
print("Neural Tags",spacy_tags)
